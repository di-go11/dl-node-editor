cmake_minimum_required(VERSION 3.18)
project(AIEngineBackend)

# C++のバージョンを 17 以上に設定（LibTorchの必須条件）
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# libtorchのパスを指定する（Dockerfileで指定したパスを使用）
set(CMAKE_PREFIX_PATH "/opt/libtorch" ${CMAKE_PREFIX_PATH}) # Cmake_Prefix_Pathに /opt/libtorch を追加するということ。右辺に ${CMAKE_PREFIX_PATH} を追加することで、既存のパスも保持される。

# 1. 以前のDockerfileで指定した環境変数（/opt/libtorch）からLibTorchを探す
find_package(Torch REQUIRED)

# 1-2. 画像デコード(JPEG/PNG)用にOpenCVを探す(Dockerfileでlibopencv-devをインストール済み)
find_package(OpenCV REQUIRED) # user/lib内に保存しているためデフォルトで見つけることができる。

# 1-3. gRPC C++ライブラリを探す。
# Ubuntu 22.04 の libgrpc++-dev(apt)は gRPCConfig.cmake / gRPCTargets.cmake を
# 提供しない(pkg-config の .pc ファイルしか無い)ため find_package(gRPC CONFIG)
# は失敗する。pkg-config 経由で見つけ、以降のコードが参照する
# gRPC::grpc++ / gRPC::grpc_cpp_plugin と同じ名前のターゲットを自前で作る。
find_package(PkgConfig REQUIRED) # PkgConfig.cmakeというCMakeの既存のモジュールを読み込む。
pkg_check_modules(GRPCPP REQUIRED IMPORTED_TARGET grpc++) # PkgConfig.cmakeのモジュールを使用してgrpc++.pcを読み込む。
add_library(gRPC::grpc++ ALIAS PkgConfig::GRPCPP) # gRPC::grpc++という読み取り専用の別名を作成。

find_program(GRPC_CPP_PLUGIN_EXECUTABLE grpc_cpp_plugin REQUIRED) # grpc_cpp_pluginはgrpcのサービススタブ(.grpc.pb.h / .grpc.pb.cc)を生成するための外部プラグイン実行ファイル。
add_executable(gRPC::grpc_cpp_plugin IMPORTED)
set_target_properties(gRPC::grpc_cpp_plugin PROPERTIES
	IMPORTED_LOCATION ${GRPC_CPP_PLUGIN_EXECUTABLE})

# devcontainer は環境変数 CMAKE_PREFIX_PATH=/opt/libtorch が設定されており、
# find_package(Protobuf) の探索がこれを拾って libtorch 同梱の protobuf ヘッダ
# (google/protobuf/*.h)を返してしまう。これは protoc(システム, 3.12.4)が
# 生成したコードや grpc++ のヘッダが前提にしているバージョンと食い違い、
# "generated by a newer version of protoc" 等のエラーになる。
# CMakeLists 内で unset(ENV{CMAKE_PREFIX_PATH}) しても CMake は起動時に
# 環境変数を内部に取り込み済みで効かないため、システムの protobuf の場所を
# 直接キャッシュに入れて探索そのものを回避する。
set(Protobuf_INCLUDE_DIR /usr/include CACHE PATH "" FORCE)
set(Protobuf_LIBRARY /usr/lib/x86_64-linux-gnu/libprotobuf.so CACHE FILEPATH "" FORCE)
set(Protobuf_PROTOC_EXECUTABLE /usr/bin/protoc CACHE FILEPATH "" FORCE)
find_package(Protobuf REQUIRED)

# find_package(Torch) が内部で読み込む Caffe2Config.cmake は、
# 「protobuf::libprotobuf という名前がまだ無ければ、中身の無いダミー
# (INTERFACE ライブラリ)を作って ALIAS する」という処理を先にやっている
# (Caffe2Targets.cmake 内の参照を解決するためだけの措置)。このため、後から
# 実行する find_package(Protobuf) は「その名前は既にある」と見なし、本物の
# ライブラリには差し替えない(FindProtobuf.cmake 側にも同種のガードがある)。
# 結果、protobuf::libprotobuf にリンクしても実体が空で、最終リンクで
# google::protobuf::Message 等が undefined reference になる。
# protobuf::libprotobuf という名前は諦め、別名で本物を用意する。
pkg_check_modules(PROTOBUF REQUIRED IMPORTED_TARGET protobuf)
add_library(SystemProtobuf::libprotobuf ALIAS PkgConfig::PROTOBUF)

# 共有契約となる proto のパスを PROTO_DIR に設定する（2 サービス共通のため ai-engine/ の外）。
cmake_path(SET PROTO_DIR NORMALIZE "${CMAKE_CURRENT_SOURCE_DIR}/../proto")

file(GLOB_RECURSE ALL_SRCS "src/training/*.cpp")

# training_smoke_test を走らせる device。CI/ローカルで -DSMOKE_DEVICE=cuda:0 と
# 指定すると GPU で実行する。既定は cpu(GPU ランナー不要)。
if(NOT DEFINED SMOKE_DEVICE)
	set(SMOKE_DEVICE "cpu")
endif()

# 2. ライブラリを作成(adapter/application/entities/infrastructure配下の実装をまとめる)
add_library(ai_engine_lib
	${ALL_SRCS}
)

# ヘッダーファイルを探す「基準の場所」を src フォルダに設定する
target_include_directories(ai_engine_lib PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)

# 3. LibTorch/OpenCVのインクルードパスとライブラリをターゲットに設定する
target_include_directories(ai_engine_lib PRIVATE
	${CMAKE_CURRENT_SOURCE_DIR}
	${TORCH_INCLUDE_DIRS}
	${OpenCV_INCLUDE_DIRS}
)
target_link_libraries(ai_engine_lib PRIVATE "${TORCH_LIBRARIES}" ${OpenCV_LIBS})

# 4. (重要) CMakeの仕様で必要なコンパイルフラグを追加
set_property(TARGET ai_engine_lib PROPERTY CXX_STANDARD 17)

# 4-2. gRPC / protobuf 生成コード専用ライブラリ。
#      .proto を「入力ソース」として渡すのが protobuf_generate の作法（configure 時に実在必須）。
#      生成物は build/ 直下だと CMakeCache 等と混ざるため、専用サブディレクトリ
#      (protoGenerated/)に集約し、.proto から生成されたコードだと分かるようにする。
set(PROTO_GEN_DIR ${CMAKE_CURRENT_BINARY_DIR}/protoGenerated)
file(MAKE_DIRECTORY ${PROTO_GEN_DIR})

add_library(ai_engine_proto STATIC
		${PROTO_DIR}/capabilities.proto
		${PROTO_DIR}/training.proto
		${PROTO_DIR}/model_graph.proto)
target_link_libraries(ai_engine_proto PUBLIC gRPC::grpc++ SystemProtobuf::libprotobuf)
target_include_directories(ai_engine_proto PUBLIC ${PROTO_GEN_DIR})
set_property(TARGET ai_engine_proto PROPERTY CXX_STANDARD 17)

# APPEND_PATH: proto ファイルが ai-engine/ の外(../proto)にあるため、これを
# 付けないと CMake が期待する生成物パスの計算(CMAKE_CURRENT_SOURCE_DIR からの
# 相対ディレクトリを PROTOC_OUT_DIR に付け足す)と、protoc が実際に書き出す
# パス(-I で渡したディレクトリを基準にしたフラットな配置)がズレて
# 「生成はされたのに add_library が探す場所に無い」状態になる。
# APPEND_PATH を付けると CMake 側もフラットな配置を期待するようになり一致する。

# (a) メッセージ型: *.pb.h / *.pb.cc
protobuf_generate(
	TARGET ai_engine_proto
	LANGUAGE cpp
	APPEND_PATH
	IMPORT_DIRS ${PROTO_DIR}
	PROTOC_OUT_DIR ${PROTO_GEN_DIR})

# (b) サービススタブ: *.grpc.pb.h / *.grpc.pb.cc（gRPC プラグイン経由）
protobuf_generate(
	TARGET ai_engine_proto
	LANGUAGE grpc
	APPEND_PATH
	GENERATE_EXTENSIONS .grpc.pb.h .grpc.pb.cc
	PLUGIN "protoc-gen-grpc=$<TARGET_FILE:gRPC::grpc_cpp_plugin>"
	IMPORT_DIRS ${PROTO_DIR}
	PROTOC_OUT_DIR ${PROTO_GEN_DIR})

# 4-3. gRPC inbound adapter（src/api/grpc/）。生成コードに触れてよいのはこの層だけ。
#
# 重要: ai_engine_lib(torch 一式)にはリンクしない。CapabilitiesServiceImpl は
# ICapabilityProvider(純粋仮想インターフェース、entities/application のみ依存)
# しか知らず、CudaCapabilityProvider のような具象 infrastructure 実装は
# GrpcApp(合成層)が組み立てて注入する(DI)。これは clean-arch 的に正しいだけでなく、
# ビルド上も理由がある: ai_engine_lib は STATIC ライブラリで torch を PRIVATE で
# リンクしているが、CMake の仕様上 STATIC ライブラリの PRIVATE 依存の使用要件
# (include dirs 等)はそれをリンクする側にも伝播してしまう(静的リンクは最終的な
# 実行ファイルが全ての依存を知る必要があるための既知の制約)。ai_engine_lib に
# リンクすると libtorch 同梱の(protoc が生成したコードと非互換な)protobuf ヘッダが
# -isystem で紛れ込み、<google/protobuf/...> の解決を奪ってしまう。
# リンクしなければこの問題自体が起きない。
#
# GrpcApp.cpp(CudaCapabilityProvider + CapabilitiesServiceImpl + GrpcServer の
# 組み立て)もここに置く。CudaCapabilityProvider.hpp 自体は torch 非依存の
# ヘッダ(メンバも無い)なので、ai_engine_lib をリンクしなくてもコンパイルできる。
# 実体(CudaCapabilityProvider.cpp、torch 使用)のリンク解決は、両方をリンクする
# 最終実行ファイル ai_engine_app の段階で行われる。
add_library(ai_engine_grpc STATIC
	src/api/grpc/GrpcServer.cpp
	src/api/grpc/CapabilitiesServiceImpl.cpp
	src/api/grpc/TrainingServiceImpl.cpp
	src/api/grpc/TrainingProtoMapper.cpp
	src/api/grpc/GrpcApp.cpp)
target_include_directories(ai_engine_grpc PRIVATE
	${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(ai_engine_grpc PRIVATE
	ai_engine_proto)
set_property(TARGET ai_engine_grpc PROPERTY CXX_STANDARD 17)

# 5. main.cppを実行可能ファイルとしてビルドし、上記ライブラリをリンクする
#
# src/main.cpp は「呼ぶだけ」の合成層で、torch のヘッダも生成コードも直接
# include しない。よって ${TORCH_INCLUDE_DIRS} は意図的にここへ付けない。
add_executable(ai_engine_app src/main.cpp)
target_include_directories(ai_engine_app PRIVATE
	${CMAKE_CURRENT_SOURCE_DIR}/src
	${CMAKE_CURRENT_SOURCE_DIR}
)
# リンク順は依存の向き通りに: GrpcApp(ai_engine_grpc) が CudaCapabilityProvider
# (ai_engine_lib) のシンボルを要求するので、要求する側を先に置く。静的リンクは
# 左から右へ一方向にしか未解決シンボルを探さないため、逆順だと
# "undefined reference to vtable for CudaCapabilityProvider" になる。
target_link_libraries(ai_engine_app PRIVATE ai_engine_grpc ai_engine_lib ai_engine_proto "${TORCH_LIBRARIES}")
set_property(TARGET ai_engine_app PROPERTY CXX_STANDARD 17)

# libtorch(CUDAビルド)が依存するcuSPARSELt/NVSHMEM等、この環境に入っていない
# オプションのCUDAライブラリのシンボル解決をリンク時ではなく実行時まで遅延させる。
# これらは今回のような疎通確認コード(スパース演算・マルチGPU通信を使わない)では
# 実行時にも参照されないため安全。
target_link_options(ai_engine_app PRIVATE "-Wl,--allow-shlib-undefined")
